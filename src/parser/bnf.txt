; GW-BASIC LL(1) Grammar
; Token names match lexer output (see src/lexer/README.md)
;
; Conventions:
;   UPPER_CASE  = terminal (token from lexer)
;   CamelCase   = nonterminal
;   ::=         = production rule
;   |           = alternative (separate lines for LL(1) clarity)
;   e           = epsilon (empty production)
;   NUM         = any NUM_i.../NUM_s.../NUM_d... token
;   VAR         = any VAR_UNK_/VAR_STR_/VAR_INT_/VAR_SNG_/VAR_DBL_ token
;   STR         = any STR_... token
;   REM_TEXT    = any REM_... token

; ============================================================
; Entry point
; ============================================================
;
; The parser always operates on a SINGLE LINE, already tokenized.
; Line numbers (LN__) are handled by the dispatcher before parsing:
;   - LN__ present → store tokenized line in program memory
;   - LN__ absent  → immediate mode: parse + execute
; The parser never sees LN__ tokens.

StmtList      ::= Stmt StmtRest
StmtRest      ::= COLON Stmt StmtRest
StmtRest      ::= e

; ============================================================
; Statements
; ============================================================

Stmt          ::= PrintStmt
Stmt          ::= LetStmt
Stmt          ::= IfStmt
Stmt          ::= ForStmt

Stmt          ::= ReadStmt

Stmt          ::= DimStmt

Stmt          ::= WEND @WEND
Stmt          ::= RemStmt
Stmt          ::= EndStmt
Stmt          ::= STOP @STOP
Stmt          ::= OnStmt
Stmt          ::= DefStmt
Stmt          ::= DefTypeStmt
Stmt          ::= PokeStmt
Stmt          ::= OpenStmt
Stmt          ::= CloseStmt
Stmt          ::= WriteStmt

Stmt          ::= LocateStmt
Stmt          ::= ColorStmt
Stmt          ::= ScreenStmt

Stmt          ::= KeyStmt

; --- PRINT / LPRINT / ? ---
;   Trailing `;` or `,` suppresses the final newline.  @PSEMI_END /
;   @PTAB_END print the just-pushed item like @PSEMI / @PTAB but do not
;   add a newline; @PEND is only emitted when the print list ends without
;   a separator.
;   Juxtaposed items (PRINT "X"N"Y") are legal GW-BASIC: adjacent items
;   with no separator behave like `;`.  The PrintAfter @PSEMI production
;   fires on any token that can START an item; tokens that can CONTINUE
;   the current expression (e.g. MINUS) are consumed by the expression
;   first, so `PRINT A -5` is still a subtraction.  This intentionally
;   adds two FIRST-vs-FOLLOW conflicts resolved in GW-BASIC's favour:
;   ArrayIndex.OPAR (`PRINT A(5)` = array access) and AddRest.MINUS.
;   Empty print zones: a COMA with no pending item emits @PZONE, which
;   advances to the next 14-column zone without popping the stack.
;   LPRINT is accepted as a console-printing alias of PRINT.
;   PRINT USING fmt$; v1; v2 ... formats values against a format string.
;   Factor through PrintWhat so USING doesn't clash with the PrintBody
;   FIRST set.  Postfix: <fmt> PU_MARK <v1> <v2> ... PRINT_USING [PU_NL].
;   PRINT_USING renders (no newline); PU_NL adds the newline when the
;   value list does not end in a trailing ; or , separator.
;   PRINT #n, ... directs output to file channel n (@PFILE_SET sets the
;   sink, @PFILE_CLR restores the console).  The reused PrintWhat machinery
;   (PEND/PSEMI/PTAB/USING) appends to the file when the sink is active.
PrintStmt     ::= PRINT PrintTarget
PrintStmt     ::= LPRINT PrintWhat
PrintTarget   ::= HASH Expr COMA @PFILE_SET PrintWhat @PFILE_CLR
PrintTarget   ::= PrintWhat
PrintWhat     ::= USING Expr SEMICOLON @PU_MARK PuList
PrintWhat     ::= PrintBody
PuList        ::= Expr PuMore
PuMore        ::= SEMICOLON PuTail
PuMore        ::= COMA PuTail
PuMore        ::= @PRINT_USING @PU_NL
PuTail        ::= Expr PuMore
PuTail        ::= @PRINT_USING
PrintBody     ::= @PEND
PrintBody     ::= PrintItem PrintAfter
PrintBody     ::= COMA @PZONE PrintZone
PrintAfter    ::= @PEND
PrintAfter    ::= SEMICOLON PrintSemi
PrintAfter    ::= COMA PrintComa
PrintAfter    ::= @PSEMI PrintItem PrintAfter
PrintSemi     ::= @PSEMI PrintItem PrintAfter
PrintSemi     ::= @PSEMI_END
PrintComa     ::= @PTAB PrintItem PrintAfter
PrintComa     ::= @PTAB COMA @PZONE PrintZone
PrintComa     ::= @PTAB_END
PrintZone     ::= PrintItem PrintAfter
PrintZone     ::= COMA @PZONE PrintZone
PrintZone     ::= e
PrintItem     ::= Expr
PrintItem     ::= TAB( Expr CPAR @FN_TAB
PrintItem     ::= SPC( Expr CPAR @FN_SPC

; --- LET / assignment ---
LetStmt       ::= LET AssignStmt
AssignStmt    ::= VAR AssignTail
AssignTail    ::= EQ Expr @ASSIGN
AssignTail    ::= @ARR_START OPAR ExprList CPAR EQ Expr @ASSIGN_ARR

; --- IF / THEN ---
; GW-BASIC accepts both `IF expr THEN ...` and `IF expr GOTO num` (a common
; shorthand from the period — equivalent to `IF expr THEN GOTO num`).
IfStmt        ::= IF Expr @IF IfBody @ENDIF
IfBody        ::= THEN ThenClause ElseClause
IfBody        ::= GOTO NUM @IF_GOTO StmtRest ElseClause
ThenClause    ::= NUM @IF_GOTO StmtRest
ThenClause    ::= StmtList
ElseClause    ::= ELSE @ELSE ThenClause
ElseClause    ::= e

; --- FOR / NEXT ---
;   @FOR_DEFAULT_STEP pushes i0001 onto the stack so @FOR always sees a step
;   value (whether the user wrote STEP or not).
ForStmt       ::= FOR VAR EQ Expr TO Expr StepClause @FOR
StepClause    ::= STEP Expr
StepClause    ::= @FOR_DEFAULT_STEP
NextStmt      ::= NEXT NextVars @NEXT
NextVars      ::= VAR NextVarRest
NextVars      ::= e
NextVarRest   ::= COMA VAR @NEXT NextVarRest
NextVarRest   ::= e

; --- GOTO / GOSUB ---
GotoStmt      ::= GOTO NUM @GOTO
GosubStmt     ::= GOSUB NUM @GOSUB

; --- ON ---
;   ON ERROR GOTO line arms a runtime-error handler (GOTO 0 disarms); the
;   error then jumps to `line` with ERR/ERL set, and RESUME returns control.
OnStmt        ::= ON OnTail
OnTail        ::= ERROR GOTO NUM @ON_ERROR
OnTail        ::= KEY OPAR Expr CPAR GOSUB NUM @ON_KEY
OnTail        ::= Expr OnAction NumList @ON_END
OnAction      ::= GOTO @ON_GOTO
OnAction      ::= GOSUB @ON_GOSUB
NumList       ::= NUM NumListRest
NumListRest   ::= COMA NUM NumListRest
NumListRest   ::= e

; --- INPUT ---
;   @INPUT_MARK pushes a sentinel between the prompt and the var list so
;   @INPUT knows where the var list starts when popping the stack.
;   `INPUT "p"; V` prints the prompt then "? "; `INPUT "p", V` prints the
;   prompt with NO question mark (@PROMPT_NQ).  Bare INPUT prints "? ".
;   INPUT #n, var,... reads from file channel n (@IFILE_SET sets _input_fh,
;   which INPUT.bat reads from instead of the console); the var-list parsing
;   and assignment are shared with console INPUT.
InputStmt     ::= INPUT InputSource
;   A leading `;` (INPUT;"p",v) suppresses the carriage return after the user's
;   entry — a cursor nuance with no effect on the value, so we just consume it.
InputSource   ::= SEMICOLON InputBody
InputSource   ::= InputBody
InputBody     ::= HASH Expr COMA @IFILE_SET @INPUT_MARK VarList @INPUT
InputBody     ::= InputPrompt @INPUT_MARK VarList @INPUT
InputPrompt   ::= STR PromptSep
PromptSep     ::= SEMICOLON @PROMPT
PromptSep     ::= COMA @PROMPT_NQ
InputPrompt   ::= e
;   VarList targets are L-values (READ/INPUT assign INTO them; DEF FN
;   params are bare names).  Array elements use @AREF, which packs the
;   array name + computed indices into a single stack token (AREF:NAME:i1:..)
;   that @READ / @INPUT store into — unlike @AIDX, which reads the element's
;   value (right for expressions, wrong for an assignment target).
VarList       ::= LValue VarListRest
VarListRest   ::= COMA LValue VarListRest
VarListRest   ::= e
LValue        ::= VAR LVIndex
LVIndex       ::= @ARR_START OPAR ExprList CPAR @AREF
LVIndex       ::= e

; --- READ / DATA / RESTORE ---
;   @READ_MARK and @DATA_MARK push stack sentinels so @READ and @DATA know
;   where the var list / value list starts when popping the stack.
;   DATA items themselves are collected at RUN time by a pre-scan of
;   program.dat (see runProgram); the runtime @DATA action just drains
;   the stack of the values that were pushed for parsing purposes.
ReadStmt      ::= READ @READ_MARK VarList @READ
DataStmt      ::= DATA @DATA_MARK DataList @DATA
DataList      ::= DataItem DataListRest
DataItem      ::= NUM
DataItem      ::= STR
DataItem      ::= MINUS NUM
DataListRest  ::= COMA DataItem DataListRest
DataListRest  ::= e
RestoreStmt   ::= RESTORE RestoreArg
RestoreArg    ::= NUM @RESTORE_LINE
RestoreArg    ::= @RESTORE

; --- DIM ---
DimStmt       ::= DIM DimVar DimVarRest
DimVar        ::= VAR @ARR_START OPAR ExprList CPAR @DIM
DimVarRest    ::= COMA DimVar DimVarRest
DimVarRest    ::= e

; --- WHILE ---
WhileStmt     ::= WHILE Expr @WHILE

; --- REM / ' ---
;   REM REM_TEXT  comes from "REM body" - explicit keyword form.
;   REM_TEXT      comes from "'body"   - apostrophe shorthand, lexer
;                                        emits the body token only.
RemStmt       ::= REM REM_TEXT @REM
RemStmt       ::= REM_TEXT @REM

; --- END ---
EndStmt       ::= END @END

; --- DEF FN ---
DefStmt       ::= DEF DefWhat
DefWhat       ::= FN VAR @PARAM_MARK OPAR DefParams CPAR EQ @DEFFN_CAP Expr @DEF_FN
DefParams     ::= VAR DefParamRest
DefParamRest  ::= COMA VAR DefParamRest
DefParamRest  ::= e
DefTypeStmt   ::= DEFINT LetterRange @DEFINT DefRestI
DefTypeStmt   ::= DEFSNG LetterRange @DEFSNG DefRestS
DefTypeStmt   ::= DEFDBL LetterRange @DEFDBL DefRestD
DefTypeStmt   ::= DEFSTR LetterRange @DEFSTR DefRestT
DefRestI      ::= COMA LetterRange @DEFINT DefRestI
DefRestI      ::= e
DefRestS      ::= COMA LetterRange @DEFSNG DefRestS
DefRestS      ::= e
DefRestD      ::= COMA LetterRange @DEFDBL DefRestD
DefRestD      ::= e
DefRestT      ::= COMA LetterRange @DEFSTR DefRestT
DefRestT      ::= e
LetterRange   ::= VAR RangeRest
RangeRest     ::= MINUS VAR
RangeRest     ::= e

; --- RUN / SAVE / SYSTEM / NEW ---
;   RUN [num | "file"[,R]] restarts the current program (clearing
;   variables), starts at a line, or loads+runs another file.  The RTLs
;   signal the run loop with flow code 98 (see exec.bat).  SAVE
;   "file"[,A|,P] writes the program (binary by default, ASCII with ,A).
;   The ,R/,A/,P option letters arrive on the stack as VAR_ tokens; the
;   RTLs discard them.
RunStmt       ::= RUN RunArg
RunArg        ::= Expr RunROpt @RUN_FILE
RunArg        ::= NUM @RUN_LINE
RunArg        ::= @RUN
RunROpt       ::= COMA VAR @RUN_ROPT
RunROpt       ::= e
SaveStmt      ::= SAVE Expr SaveOpt @SAVE
SaveOpt       ::= COMA VAR
SaveOpt       ::= e

; --- POKE ---
PokeStmt      ::= POKE Expr COMA Expr @POKE

;   OPEN has three real-world shapes (see io-gap-analysis):
;     A)  OPEN modeExpr , [#]chan , fileExpr [, reclen]   mode = "O"/"I"/"A"/"R"
;     B)  OPEN fileExpr FOR access AS [#]chan             sequential
;     C)  OPEN fileExpr AS [#]chan [LEN= reclen]          random
;   The first Expr is the mode (A) or the filename (B/C); the next token
;   (COMA / FOR / AS) selects the form.  @ONIL marks an omitted reclen/LEN.
OpenStmt      ::= OPEN Expr OpenTail
OpenTail      ::= COMA OpenChan COMA Expr OpenRecLen @OPEN_M
OpenTail      ::= FOR OpenFor
OpenTail      ::= AS OpenChan OpenLen @OPEN_R
OpenFor       ::= INPUT AS OpenChan @OPEN_FI
OpenFor       ::= OUTPUT AS OpenChan @OPEN_FO
OpenFor       ::= APPEND AS OpenChan @OPEN_FA
OpenChan      ::= HASH Expr
OpenChan      ::= Expr
OpenRecLen    ::= COMA Expr
OpenRecLen    ::= @ONIL
OpenLen       ::= LEN EQ Expr
OpenLen       ::= @ONIL

; --- CLOSE ---
;   CLOSE (bare = all), CLOSE #n, CLOSE n, CLOSE #1,#2 ...
CloseStmt     ::= CLOSE @CLOSE_MARK CloseList @CLOSE
CloseList     ::= CloseItem CloseRest
CloseList     ::= e
CloseItem     ::= HASH Expr
CloseItem     ::= Expr
CloseRest     ::= COMA CloseItem CloseRest
CloseRest     ::= e

; --- WRITE ---
;   WRITE #n, v1, v2 ...  writes comma-separated values (strings quoted)
;   plus CRLF to channel n; bare WRITE does the same to the console.
WriteStmt     ::= WRITE WriteHead @WRITE_MARK ExprList @WRITE
WriteHead     ::= HASH Expr COMA @PFILE_SET
WriteHead     ::= e

; --- Directory / file management (FILES / KILL / NAME / RESET) ---
;   FILES has an optional filespec; the two FilesArg productions dispatch
;   to @FILES (a spec is on the stack) or @FILES_ALL (list the cwd).  KILL
;   and RESET are inline in Stmt.  NAME old AS new renames.
FilesStmt     ::= FILES FilesArg
FilesArg      ::= Expr @FILES
FilesArg      ::= @FILES_ALL
NameStmt      ::= NAME Expr AS Expr @NAME

; --- WIDTH (screen/device column width) — accepted, no-op (we don't model a
;   fixed terminal width).  Forms: WIDTH n / WIDTH n,m / WIDTH "dev",n .
WidthStmt     ::= WIDTH @WIDTH_MARK Expr WidthRest @WIDTH
WidthRest     ::= COMA Expr WidthRest
WidthRest     ::= e

; --- CLEAR — reset all variables/arrays and close files (program kept).
;   Optional numeric args are memory/stack sizes; accepted and ignored.
;   Forms: CLEAR / CLEAR n / CLEAR ,n / CLEAR ,,n / CLEAR n,m .
ClearStmt     ::= CLEAR @CLEAR_MARK ClearArgs @CLEAR
ClearArgs     ::= Expr ClearRest
ClearArgs     ::= COMA ClearOpt ClearRest
ClearArgs     ::= e
ClearRest     ::= COMA ClearOpt ClearRest
ClearRest     ::= e
ClearOpt      ::= Expr
ClearOpt      ::= e

; --- CHAIN [MERGE] file[, [line][, ALL][, DELETE lo-hi]] ---
;   Loads/overlays another program, PRESERVING variables (unlike RUN).  The
;   filename and the optional start line are pushed; ALL is accepted (we always
;   preserve all variables); @CHAIN_DEL captures a MERGE delete-range; the line
;   slot defaults to @CHAIN_NOLINE (CHAIN_NIL) so @CHAIN always sees a line.
ChainStmt     ::= CHAIN ChainKind
ChainKind     ::= MERGE Expr ChainLine @CHAIN_MERGE
ChainKind     ::= Expr ChainLine @CHAIN
ChainLine     ::= COMA ChainLineVal ChainSkip
ChainLine     ::= @CHAIN_NOLINE
ChainLineVal  ::= Expr
ChainLineVal  ::= @CHAIN_NOLINE
ChainSkip     ::= COMA ChainSkip
ChainSkip     ::= ALL ChainSkip
ChainSkip     ::= DELETE NUM MINUS NUM @CHAIN_DEL ChainSkip
ChainSkip     ::= e

; --- LIST [n][-][m] : list program lines (n-m / n- / -m / n / all).  Like
;   GW-BASIC, listing returns to command level, so @LIST halts (flow 99).
;   Omitted start -> LIST_MIN sentinel (line 0); omitted end -> LIST_MAX.
ListStmt      ::= LIST ListArgs @LIST
ListArgs      ::= NUM ListAfterNum
ListArgs      ::= MINUS @LIST_PUSH0 NUM
ListArgs      ::= @LIST_AE
ListAfterNum  ::= MINUS ListEnd
ListAfterNum  ::= @LIST_DUP
ListEnd       ::= NUM
ListEnd       ::= @LIST_MAX

; --- MID$ statement (left-side substring assignment) ---
;   MID$(target$, start [, len]) = value  overwrites up to `len` chars of
;   target$ starting at `start` (1-based) with `value`.
MidStmt       ::= MID$ OPAR LValue COMA Expr MidSetLen CPAR EQ Expr @MID_SET
MidSetLen     ::= COMA Expr
MidSetLen     ::= @MID_DEFAULT_LEN

; --- Random-access records: FIELD / GET / PUT / LSET / RSET ---
;   FIELD [#]n, w1 AS v1$, w2 AS v2$, ... declares field vars as live windows
;   over handle n's record buffer (Architecture A — see io-gap-analysis).
;   Targets are L-values (subscripted field vars FILL$(I) use @AREF).  GET/PUT
;   move the buffer to/from record (rec-1)*reclen; LSET/RSET pack into it.
FieldStmt     ::= FIELD FileChan COMA @FIELD_MARK FieldList @FIELD
FileChan      ::= HASH Expr
FileChan      ::= Expr
FieldList     ::= Expr AS LValue FieldRest
FieldRest     ::= COMA Expr AS LValue FieldRest
FieldRest     ::= e
GetStmt       ::= GET FileChan RecOpt @GET
PutStmt       ::= PUT FileChan RecOpt @PUT
RecOpt        ::= COMA Expr
RecOpt        ::= @ONIL
LsetStmt      ::= LSET LValue EQ Expr @LSET
RsetStmt      ::= RSET LValue EQ Expr @RSET

; --- LINE ---
LineStmt      ::= LINE LineWhat
LineWhat      ::= INPUT LineSource
LineSource    ::= HASH Expr COMA @IFILE_SET VarRef @LINE_INPUT
LineSource    ::= LineInput @LINE_INPUT
LineWhat      ::= OPAR Expr COMA Expr CPAR MINUS OPAR Expr COMA Expr CPAR LineBox @LINE_DRAW
LineInput     ::= InputPrompt VarRef
LineBox       ::= COMA Expr LineBoxFill
LineBox       ::= e
LineBoxFill   ::= COMA VAR
LineBoxFill   ::= e

; --- LOCATE ---
;   LOCATE [row][,[col][,[cursor][,[start][,stop]]]] — every argument is
;   optional and may be elided with a bare comma (LOCATE ,52 / LOCATE ,,0).
;   @LOC_MARK pushes a sentinel; each slot pushes its value or a LOC_NIL
;   sentinel (@LOC_NIL for an elided arg); @LOCATE reads them positionally
;   (row, col, cursor, start, stop) and emits ANSI cursor control.
LocateStmt    ::= LOCATE @LOC_MARK LocArgs @LOCATE
LocArgs       ::= LocArg LocMore
LocArg        ::= Expr
LocArg        ::= @LOC_NIL
LocMore       ::= COMA LocArg LocMore
LocMore       ::= e

; --- COLOR ---
;   COLOR [fg][,bg][,border] in text mode.  Same elided-arg protocol as
;   LOCATE: @COL_MARK sentinel, each slot pushes its value or COL_NIL.
;   @COLOR maps CGA colour numbers to ANSI SGR (border is ignored).
ColorStmt     ::= COLOR @COL_MARK ColArgs @COLOR
ColArgs       ::= ColArg ColMore
ColArg        ::= Expr
ColArg        ::= @COL_NIL
ColMore       ::= COMA ColArg ColMore
ColMore       ::= e

; --- SCREEN ---
ScreenStmt    ::= SCREEN Expr ScreenRest @SCREEN
ScreenRest    ::= COMA Expr ScreenRest2
ScreenRest    ::= e
ScreenRest2   ::= COMA Expr ScreenRest3
ScreenRest2   ::= e
ScreenRest3   ::= COMA Expr
ScreenRest3   ::= e

; --- SOUND ---
SoundStmt     ::= SOUND Expr COMA Expr @SOUND

; --- KEY ---
KeyStmt       ::= KEY KeyWhat
KeyWhat       ::= ON @KEY_ON
KeyWhat       ::= OFF @KEY_OFF
KeyWhat       ::= LIST @KEY_LIST
; KEY(n) ON/OFF/STOP arms function-key event trapping.  The trap form is
; listed AFTER the KEY n,str$ soft-key form on purpose: the table builder lets
; the later rule win a FIRST/FIRST conflict, so `KEY (` resolves to trapping
; rather than a parenthesised KEY_SET first arg (which the corpus never uses).
KeyWhat       ::= Expr COMA Expr @KEY_SET
KeyWhat       ::= OPAR Expr CPAR KeyTrap
KeyTrap       ::= ON @KEY_TRAP
KeyTrap       ::= OFF @KEY_TRAP
KeyTrap       ::= STOP @KEY_TRAP

; --- SWAP ---
SwapStmt      ::= SWAP LValue COMA LValue @SWAP

; --- ERASE ---
EraseStmt     ::= ERASE VAR @ERASE EraseRest
EraseRest     ::= COMA VAR @ERASE EraseRest
EraseRest     ::= e

; --- ERROR / RESUME ---
ErrorStmt     ::= ERROR Expr @ERROR
ResumeStmt    ::= RESUME ResumeTarget
ResumeTarget  ::= NEXT @RESUME_NEXT
ResumeTarget  ::= NUM @RESUME_LINE
ResumeTarget  ::= @RESUME

; ============================================================
; Expressions (precedence low to high, no left recursion)
; ============================================================
;
; Precedence (GW-BASIC, lowest first):
;   1. IMP
;   2. EQV
;   3. XOR
;   4. OR
;   5. AND
;   6. NOT  (prefix)
;   7. =  <  >  <=  >=  <>   (relational)
;   8. +  -  (additive)
;   9. MOD
;  10. \  (integer division)
;  11. *  /  (multiplicative)
;  12. -  (unary negation)
;  13. ^  (exponentiation, right-assoc)
;  14. primary

Expr          ::= ImpExpr

ImpExpr       ::= EqvExpr ImpRest
ImpRest       ::= IMP EqvExpr @IMP ImpRest
ImpRest       ::= e

EqvExpr       ::= XorExpr EqvRest
EqvRest       ::= EQV XorExpr @EQV EqvRest
EqvRest       ::= e

XorExpr       ::= OrExpr XorRest
XorRest       ::= XOR OrExpr @XOR XorRest
XorRest       ::= e

OrExpr        ::= AndExpr OrRest
OrRest        ::= OR AndExpr @OR OrRest
OrRest        ::= e

AndExpr       ::= NotExpr AndRest
AndRest       ::= AND NotExpr @AND AndRest
AndRest       ::= e

NotExpr       ::= NOT NotExpr @NOT
NotExpr       ::= RelExpr

RelExpr       ::= AddExpr RelRest
RelRest       ::= EQ AddExpr @CMP_EQ RelRest
RelRest       ::= LT AddExpr @CMP_LT RelRest
RelRest       ::= GT AddExpr @CMP_GT RelRest
RelRest       ::= LE AddExpr @CMP_LE RelRest
RelRest       ::= GE AddExpr @CMP_GE RelRest
RelRest       ::= NE AddExpr @CMP_NE RelRest
RelRest       ::= e

AddExpr       ::= ModExpr AddRest
AddRest       ::= PLUS ModExpr @ADD AddRest
AddRest       ::= MINUS ModExpr @SUB AddRest
AddRest       ::= e

ModExpr       ::= IdivExpr ModRest
ModRest       ::= MOD IdivExpr @MOD ModRest
ModRest       ::= e

IdivExpr      ::= MulExpr IdivRest
IdivRest      ::= IDIV MulExpr @IDIV IdivRest
IdivRest      ::= e

MulExpr       ::= UnaryExpr MulRest
MulRest       ::= MUL UnaryExpr @MUL MulRest
MulRest       ::= DIV UnaryExpr @DIV MulRest
MulRest       ::= e

UnaryExpr     ::= MINUS PowExpr @NEG
UnaryExpr     ::= PowExpr

PowExpr       ::= Primary PowRest
PowRest       ::= POW PowExpr @POW
PowRest       ::= e

BuiltinCall   ::= LogFn

BuiltinCall   ::= CintFn
BuiltinCall   ::= CsngFn
BuiltinCall   ::= CdblFn

BuiltinCall   ::= ValFn
BuiltinCall   ::= AscFn
BuiltinCall   ::= PeekFn
BuiltinCall   ::= InpFn
BuiltinCall   ::= PosFn
BuiltinCall   ::= FreFn
BuiltinCall   ::= MkiFn
BuiltinCall   ::= MksFn

; RND can be RND or RND(x)
RndFn         ::= RND RndArgs @FN_RND

; Bare `RND` (no parens) is equivalent to RND(1) — push a default arg.
RndArgs       ::= @RND_NOARG

; 2-arg string functions

RightFn       ::= RIGHT$ OPAR Expr COMA Expr CPAR @FN_RIGHT

; 2-3 arg
;   @MID_DEFAULT_LEN pushes a sentinel large length so @FN_MID always sees
;   three values on the stack.
MidFn         ::= MID$ OPAR Expr COMA Expr MidLen CPAR @FN_MID
MidLen        ::= COMA Expr
MidLen        ::= @MID_DEFAULT_LEN

; 2-3 arg
InstrFn       ::= INSTR OPAR Expr COMA Expr InstrLen CPAR @FN_INSTR
InstrLen      ::= COMA Expr
InstrLen      ::= @INSTR_NIL
